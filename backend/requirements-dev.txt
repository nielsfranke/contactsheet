# Test-only dependencies (NOT installed into the runtime image — that uses requirements.txt).
# Install with: pip install -r requirements.txt -r requirements-dev.txt
#
# Unit suite (backend/tests/) runs under FastAPI's TestClient; the end-to-end suite (../e2e/)
# drives a real browser via Playwright against live backend + frontend servers. After installing,
# fetch the browser once with: playwright install chromium

pytest==9.1.1
# Playwright's pytest plugin brings the `page`/`context`/`browser` fixtures. The `playwright`
# wheel is a transitive dep (already present in this venv for screenshot verification).
# 0.9.0 is the floor for pytest 9 — 0.7.x capped pytest <9 and blocked the whole suite's upgrade.
pytest-playwright==0.9.0
# httpx is also a runtime dep (pinned in requirements.txt); the e2e harness uses it to poll the
# backend/frontend health endpoints while booting the servers.

# Kill a hung test (a socket left open, a thread that never joins) instead of hanging CI for 6 h.
pytest-timeout==2.3.1
# One retry for the browser-driven e2e suite only (`--reruns 1` in CI); the unit suite never retries.
pytest-rerunfailures==16.6.1
